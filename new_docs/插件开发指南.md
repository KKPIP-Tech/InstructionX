# InstructionX 插件开发指南

本文档将指导你如何为 InstructionX 开发自定义插件。通过插件机制，你可以扩展应用功能，创建自己的工具。

---

## 1. 快速入门

### 1.1 创建插件步骤

```
1. 创建插件目录
2. 编写 entrance.py（必须）
3. 编写 information.py（必须）
4. 编写 service.py（必须）
5. 创建 __init__.py（必须）
6. 重启应用，插件自动加载
```

### 1.2 最小插件示例

```python
# custom_plugin/hello_world/entrance.py
from PySide6.QtWidgets import QWidget, QVBoxLayout, QLabel
from core.plugin.plugin_interface import IPlugin

class HelloWorldPlugin(IPlugin):
    @property
    def plugin_name(self) -> str:
        return "hello_world"

    def _create_widget(self, parent=None, data_provider=None):
        widget = QWidget(parent)
        layout = QVBoxLayout()
        layout.addWidget(QLabel("你好，世界！"))
        widget.setLayout(layout)
        return widget
```

## 2. 插件目录结构

每个插件必须放在 `custom_plugin/` 目录下，遵循以下结构：

```
custom_plugin/
└── your_plugin_name/
    ├── __init__.py           # 空文件，必须存在
    ├── entrance.py           # 插件入口类（必须）
    ├── information.py        # 插件元数据（必须）
    └── service.py            # 服务层（必须）
```

### 目录说明

| 文件 | 必须 | 说明 |
|------|------|------|
| `entrance.py` | ✅ | 定义继承 IPlugin 的入口类 |
| `information.py` | ✅ | 定义插件元数据（版本、图标、API等） |
| `service.py` | ✅ | 实现服务层逻辑，提供 API |
| `__init__.py` | ✅ | 空文件，使目录成为 Python 包 |

## 3. 核心文件详解

### 3.1 entrance.py - 插件入口

`entrance.py` 是插件的入口文件，必须定义一个继承 `IPlugin` 的类。

#### 完整示例

```python
from PySide6.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout,
                                QLabel, QPushButton, QTextEdit)
from core.plugin.plugin_interface import IPlugin
from core.data.data_provider import DataProvider

class ColorConverterPlugin(IPlugin):
    """颜色转换插件"""

    def __init__(self):
        self._cached_widget = None
        self._data_provider = DataProvider()
        self._data_provider.register_plugin("color_converter", "custom")

    @property
    def plugin_name(self) -> str:
        return "color_converter"

    def _create_widget(self, parent=None, data_provider=None):
        """创建插件 UI 控件"""
        widget = QWidget(parent)
        main_layout = QVBoxLayout()

        # 标题
        title = QLabel("颜色转换器")
        title.setStyleSheet("font-size: 16px; font-weight: bold;")
        main_layout.addWidget(title)

        # 输入区域
        input_layout = QHBoxLayout()
        input_layout.addWidget(QLabel("HEX:"))
        self.hex_input = QTextEdit()
        self.hex_input.setMaximumHeight(30)
        self.hex_input.setPlaceholderText("#FF5733")
        input_layout.addWidget(self.hex_input)
        main_layout.addLayout(input_layout)

        # 转换按钮
        convert_btn = QPushButton("转换为 RGB")
        convert_btn.clicked.connect(self.convert_color)
        main_layout.addWidget(convert_btn)

        # 输出区域
        self.result_label = QLabel("RGB: -")
        main_layout.addWidget(self.result_label)

        # 订阅其他插件数据变化（如任务管理器）
        self._data_provider.subscribe(
            subscriber_id="color_converter",
            target_plugin_id="task_manager",
            target_key="tasks",
            callback=self.on_tasks_changed,
            namespace="PUBLIC"
        )

        widget.setLayout(main_layout)
        return widget

    def convert_color(self):
        """转换颜色"""
        hex_color = self.hex_input.toPlainText().strip()
        if hex_color.startswith('#'):
            hex_color = hex_color[1:]

        try:
            r = int(hex_color[0:2], 16)
            g = int(hex_color[2:4], 16)
            b = int(hex_color[4:6], 16)
            self.result_label.setText(f"RGB: ({r}, {g}, {b})")
        except (ValueError, IndexError):
            self.result_label.setText("无效的 HEX 颜色值")

    def on_tasks_changed(self, old_value, new_value):
        """当任务列表变化时的回调"""
        print(f"检测到任务变化: {len(new_value)} 个任务")
```

### 3.2 information.py - 插件元数据

`information.py` 定义插件的元数据信息，用于在技能面板中显示。

#### 完整示例

```python
from core.plugin.plugin_info_interface import IPluginInfo
from core.plugin.plugin_version import PluginVersion
from core.plugin.plugin_icon import PluginIcon
from typing import Dict

class ColorConverterInfo(IPluginInfo):
    """颜色转换插件元数据"""

    @property
    def version(self) -> PluginVersion:
        return PluginVersion(major=1, minor=0, patch=0)

    @property
    def developer(self) -> str:
        return "Your Name"

    @property
    def description(self) -> str:
        return "一款方便的颜色转换工具，支持 HEX 与 RGB 互转"

    @property
    def service_api(self) -> Dict:
        """定义插件提供的 API"""
        return {
            "methods": {
                "convert_hex_to_rgb": {
                    "description": "将 HEX 颜色转换为 RGB",
                    "parameters": {
                        "hex_color": {"type": "string", "required": True}
                    }
                },
                "convert_rgb_to_hex": {
                    "description": "将 RGB 颜色转换为 HEX",
                    "parameters": {
                        "r": {"type": "integer", "required": True},
                        "g": {"type": "integer", "required": True},
                        "b": {"type": "integer", "required": True}
                    }
                }
            }
        }

    @property
    def skill_icon(self) -> PluginIcon:
        # 从资源加载图标
        return PluginIcon.from_resource("color_converter_icon")

    @property

        return "HEX/RGB 颜色转换"
```

#### 元数据字段说明

| 字段 | 类型 | 说明 |
|------|------|------|
| `version` | PluginVersion | 插件版本号 |
| `developer` | str | 开发者名称 |
| `description` | str | 插件详细描述 |
| `service_api` | Dict | 插件提供的 API 定义 |
| `skill_icon` | PluginIcon | 技能面板图标 |
| `skill_description` | str | 技能面板简短描述 |

### 3.3 service.py - 服务层

`service.py` 必须存在，用于定义插件的业务逻辑，并暴露可被其他插件调用的 API。

#### 完整示例

```python
class Service:
    """颜色转换服务"""

    def convert_hex_to_rgb(self, hex_color: str) -> dict:
        """HEX 转 RGB"""
        hex_color = hex_color.strip()
        if hex_color.startswith('#'):
            hex_color = hex_color[1:]

        r = int(hex_color[0:2], 16)
        g = int(hex_color[2:4], 16)
        b = int(hex_color[4:6], 16)

        return {"r": r, "g": g, "b": b}

    def convert_rgb_to_hex(self, r: int, g: int, b: int) -> str:
        """RGB 转 HEX"""
        return f"#{r:02X}{g:02X}{b:02X}"
```

## 4. 插件接口详解

### 4.1 IPlugin 接口方法

```python
class IPlugin(ABC):
    @property
    @abstractmethod
    def plugin_name(self) -> str:
        """插件名称，必须唯一"""
        pass

    @abstractmethod
    def _create_widget(self, parent=None, data_provider=None) -> QWidget:
        """创建插件 UI 控件"""
        pass

    def get_widget(self, parent=None, data_provider=None) -> QWidget:
        """获取插件控件（带缓存）"""
        pass

    def get_plugin_id(self) -> str:
        """获取插件唯一 ID"""
        pass
```

### 4.2 Widget 缓存机制

`get_widget()` 方法内部实现了缓存机制：

```python
def get_widget(self, parent=None, data_provider=None):
    # 如果已有缓存且父控件相同，直接返回缓存
    if self._cached_widget is not None and self._cached_parent is parent:
        return self._cached_widget

    # 否则创建新控件
    widget = self._create_widget(parent, data_provider)

    # 缓存控件
    self._cached_widget = widget
    self._cached_parent = parent

    return widget
```

**注意**：不要在 `_create_widget()` 中缓存控件，应该让基类的 `get_widget()` 处理缓存。

## 5. 插件间通信

### 5.1 发布/订阅模式

使用 DataProvider 的发布/订阅机制实现插件间通信。

```
┌─────────────────────────────────────────────────────────────┐
│                    发布/订阅流程                             │
├─────────────────────────────────────────────────────────────┤
│                                                              │
│  插件 A (发布者)                                            │
│       │                                                     │
│       │ data_provider.publish(key, value)                  │
│       ▼                                                     │
│  ┌─────────────────────────────────────────────┐           │
│  │              DataProvider                    │           │
│  │  - 存储数据                                   │           │
│  │  - 触发订阅者的回调                           │           │
│  └──────────────────────┬──────────────────────┘           │
│                         │                                    │
│       ┌─────────────────┼─────────────────┐                  │
│       ▼                 ▼                 ▼                  │
│  回调函数 A          回调函数 B        回调函数 C            │
│  (插件 B)           (插件 C)          (插件 D)              │
│                                                              │
└─────────────────────────────────────────────────────────────┘
```

#### 发布数据

```python
# 在 entrance.py 中
from core.data.data_provider import DataProvider

data_provider = DataProvider()

# 发布数据更新
data_provider.publish(
    publisher_id="my_plugin",
    key="data_key",
    value={"some": "data"},
    namespace="PUBLIC"  # 或 "PRIVATE"
)
```

#### 订阅数据

```python
# 在 _create_widget 中订阅
data_provider.subscribe(
    subscriber_id="my_plugin",
    target_plugin_id="target_plugin",
    target_key="data_key",
    callback=self.on_data_changed,
    namespace="PUBLIC"
)

# 定义回调函数
def on_data_changed(self, old_value, new_value):
    print(f"数据从 {old_value} 变为 {new_value}")
```

## 7. 完整示例：颜色转换插件

以下是一个完整的插件示例，包含所有三个核心文件。

### 7.1 目录结构

```
custom_plugin/
└── color_converter/
    ├── entrance.py
    ├── information.py
    └── service.py
```

### 7.2 entrance.py

```python
from PySide6.QtWidgets import (QWidget, QVBoxLayout, QHBoxLayout,
                                QLabel, QPushButton, QTextEdit, QLineEdit)
from core.plugin.plugin_interface import IPlugin
from core.data.data_provider import DataProvider
from .service import Service

class ColorConverterPlugin(IPlugin):
    def __init__(self):
        self._cached_widget = None
        self._data_provider = DataProvider()
        self._data_provider.register_plugin("color_converter", "custom")
        self._service = Service()

    @property
    def plugin_name(self) -> str:
        return "color_converter"

    def _create_widget(self, parent=None, data_provider=None):
        widget = QWidget(parent)
        layout = QVBoxLayout()

        # 标题
        title = QLabel("颜色转换器")
        title.setStyleSheet("font-size: 18px; font-weight: bold;")
        layout.addWidget(title)

        # HEX 输入
        layout.addWidget(QLabel("HEX 颜色:"))
        self.hex_input = QLineEdit()
        self.hex_input.setPlaceholderText("#FF5733")
        layout.addWidget(self.hex_input)

        # RGB 输入
        layout.addWidget(QLabel("RGB 颜色:"))
        rgb_layout = QHBoxLayout()
        self.r_input = QLineEdit()
        self.r_input.setPlaceholderText("R")
        self.g_input = QLineEdit()
        self.g_input.setPlaceholderText("G")
        self.b_input = QLineEdit()
        self.b_input.setPlaceholderText("B")
        rgb_layout.addWidget(self.r_input)
        rgb_layout.addWidget(self.g_input)
        rgb_layout.addWidget(self.b_input)
        layout.addLayout(rgb_layout)

        # 按钮
        btn_layout = QHBoxLayout()
        hex_to_rgb_btn = QPushButton("HEX → RGB")
        hex_to_rgb_btn.clicked.connect(self.convert_hex_to_rgb)
        btn_layout.addWidget(hex_to_rgb_btn)

        rgb_to_hex_btn = QPushButton("RGB → HEX")
        rgb_to_hex_btn.clicked.connect(self.convert_rgb_to_hex)
        btn_layout.addWidget(rgb_to_hex_btn)
        layout.addLayout(btn_layout)

        # 结果显示
        self.result_label = QLabel("结果: -")
        self.result_label.setStyleSheet("font-size: 14px; padding: 10px;")
        layout.addWidget(self.result_label)

        layout.addStretch()
        widget.setLayout(layout)
        return widget

    def convert_hex_to_rgb(self):
        hex_color = self.hex_input.text().strip()
        if not hex_color.startswith('#'):
            hex_color = '#' + hex_color

        try:
            result = self._service.convert_hex_to_rgb(hex_color)
            self.result_label.setText(
                f"结果: RGB({result['r']}, {result['g']}, {result['b']})"
            )
            # 填充 RGB 输入框
            self.r_input.setText(str(result['r']))
            self.g_input.setText(str(result['g']))
            self.b_input.setText(str(result['b']))
        except Exception as e:
            self.result_label.setText(f"错误: {str(e)}")

    def convert_rgb_to_hex(self):
        try:
            r = int(self.r_input.text())
            g = int(self.g_input.text())
            b = int(self.b_input.text())
            result = self._service.convert_rgb_to_hex(r, g, b)
            self.result_label.setText(f"结果: {result}")
            self.hex_input.setText(result)
        except Exception as e:
            self.result_label.setText(f"错误: {str(e)}")
```

class Service:
    def convert_hex_to_rgb(self, hex_color: str) -> dict:
        """将 HEX 颜色转换为 RGB"""
        hex_color = hex_color.strip()

## 8. 最佳实践

### 8.1 命名规范

- 插件目录名：使用小写字母和下划线，如 `color_converter`
- 插件类名：使用驼峰命名，如 `ColorConverterPlugin`
- 插件名称：使用下划线命名，如 `color_converter`

### 8.2 初始化顺序

1. 在 `__init__` 中初始化 DataProvider
2. 注册插件：`data_provider.register_plugin()`
3. 在 `_create_widget` 中订阅数据变化

### 8.3 错误处理

```python
def convert_color(self):
    try:
        # 转换逻辑
        pass
    except ValueError as e:
        # 显示错误信息
        self.result_label.setText(f"错误: {str(e)}")
    except Exception as e:
        # 捕获所有异常
        print(f"未知错误: {str(e)}")
```

### 8.4 资源管理

- 图标放在 `assets/` 目录
- 使用 `PluginIcon.from_resource()` 加载图标
- 大型资源使用 `save_asset()` / `load_asset()` 管理

## 10. 总结

开发插件的核心要点：

1. **创建入口类**：继承 `IPlugin`，实现 `plugin_name` 和 `_create_widget`
2. **定义元数据**：使用 `IPluginInfo` 定义版本、图标、API
3. **实现服务层**：在 `service.py` 中提供可被调用的方法
4. **创建 __init__.py**：使目录成为 Python 包
5. **数据通信**：使用 `DataProvider` 的发布/订阅机制
6. **测试验证**：重启应用，插件将自动加载

祝你开发愉快！
